function moveImageLayer(img,where){const imgs=[...paper.querySelectorAll('image[data-type="image"]')].filter(x=>x!==img);if(!imgs.length)return;if(where==='front'){const frame=paper.querySelector('[data-type=frame]');if(frame)paper.insertBefore(img,frame);else imgs[imgs.length-1].after(img)}else{paper.insertBefore(img,imgs[0])}drawSelection();setStatus(where==='front'?'Картинка расположена поверх других картинок':'Картинка расположена под другими картинками')}
paper.addEventListener('pointerdown',e=>{const p=pt(e),obj=groupType(e.target);if(obj&&['dimension','text'].includes(obj.dataset?.type))lastEditable=obj;
 if(pasteDraft){e.preventDefault();updatePasteGhost(p);placePaste();return}
 if(tool==='dimension'){dimDraft={p1:p,p2:p};dimPreview=el('line',{x1:p.x,y1:p.y,x2:p.x,y2:p.y,stroke:'#2563eb','stroke-width':'.5','stroke-dasharray':'2 1','vector-effect':'non-scaling-stroke'});paper.appendChild(dimPreview);paper.setPointerCapture?.(e.pointerId);return}
 if(tool==='text'){textDraft={p1:p,p2:p};textPreview=el('rect',{x:p.x,y:p.y,width:0,height:0,rx:3,ry:3,class:'text-draft'});paper.appendChild(textPreview);paper.setPointerCapture?.(e.pointerId);return}
 if(e.target.classList?.contains('image-handle')){const g=selected?.dataset.type==='image'?selected:findOwner(e.target.dataset.owner,'image');if(g&&activeImage===g&&tool==='image-edit'){const r=imageRect(g);rememberImageGeometry(g);imageResize={obj:g,corner:e.target.dataset.corner,start:p,...r};e.preventDefault();return}}
 if(e.target.classList?.contains('rotate-handle')&&selected?.dataset.type==='text'){const g=selected,x=+g.dataset.x,y=+g.dataset.y,w=+g.dataset.w,h=+g.dataset.h;rotateDrag={obj:g,cx:x+w/2,cy:y+h/2,startAngle:+g.dataset.angle||0,startPointer:Math.atan2(p.y-(y+h/2),p.x-(x+w/2))*180/Math.PI};e.preventDefault();return}
 if(e.target.classList?.contains('resize-handle')&&selected?.dataset.type==='text'){textResize={obj:selected,start:p,w:+selected.dataset.w,h:+selected.dataset.h};e.preventDefault();return}
 if(e.target.classList?.contains('dim-handle-start')||e.target.classList?.contains('dim-handle-end')){const id=e.target.dataset.owner,g=selected?.dataset.type==='dimension'?selected:findOwner(id,'dimension');if(g){if(selected!==g)select(g);const end=e.target.classList.contains('dim-handle-start')?'p1':'p2',center=parsePt(g.dataset[end]);endpointDrag={obj:g,end,offX:center.x-p.x,offY:center.y-p.y};e.preventDefault();return}}
 if(e.target.classList?.contains('dim-tail-handle')){const id=e.target.dataset.owner,g=selected?.dataset.type==='dimension'?selected:findOwner(id,'dimension');if(g){if(selected!==g)select(g);tailDrag={obj:g};e.preventDefault();return}}
 if(obj?.dataset.type==='dimension'&&obj!==selected)select(obj);
 if(obj?.dataset.type==='text'&&obj!==selected){selected=obj;showProps();drawSelection()}
 if(!obj)return;
 if(e.target.classList?.contains('dim-label')&&obj.dataset.type==='dimension'){setTimeout(()=>{$('dimText').focus();$('dimText').select()},0);return}
 drag={obj,start:p};if(obj.dataset.type==='dimension'){drag.p1=parsePt(obj.dataset.p1);drag.p2=parsePt(obj.dataset.p2)}else if(obj.dataset.type==='text'){drag.x=+obj.dataset.x;drag.y=+obj.dataset.y}else if(obj.dataset.type==='image'){if(activeImage!==obj||tool!=='image-edit'){drag=null;return}rememberImageGeometry(obj);drag.x=+obj.getAttribute('x');drag.y=+obj.getAttribute('y')}
});
paper.addEventListener('contextmenu',e=>{const obj=groupType(e.target);if(obj?.dataset.type==='image'){e.preventDefault();openImageMenu(obj,e)}else closeImageMenu()});
$('imageContextMenu').addEventListener('pointerdown',e=>e.stopPropagation());$('imageContextMenu').addEventListener('click',e=>{const b=e.target.closest('button[data-layer]');if(!b)return;const img=selected?.dataset.type==='image'?selected:null;if(img)moveImageLayer(img,b.dataset.layer);closeImageMenu()});window.addEventListener('pointerdown',e=>{if(!e.target.closest('#imageContextMenu'))closeImageMenu()});
paper.addEventListener('dblclick',e=>{const obj=groupType(e.target);if(!obj)return;e.preventDefault();if(obj.dataset.type==='text'){select(obj);setTool('text-edit',true);editText(obj)}else if(obj.dataset.type==='dimension'){select(obj);setTool('dimension-edit',true);syncDimProps();openDimPopup()}else if(obj.dataset.type==='image'){drag=null;imageResize=null;activeImage=obj;select(obj);tool='image-edit';setStatus('Режим картинки включён двойным кликом: теперь её можно перемещать или менять размер. Esc — зафиксировать')}});
paper.addEventListener('pointermove',e=>{const p=pt(e);
 if(pasteDraft){updatePasteGhost(p);return}
 if(!dimDraft&&!textDraft&&!drag&&!textResize&&!imageResize&&!endpointDrag&&!tailDrag&&!rotateDrag&&tool!=='dimension'&&tool!=='text'){const hd=groupType(e.target),nd=hd?.dataset.type==='dimension'?hd:null;if(nd!==hoverDim){hoverDim=nd;if(nd)lastEditable=nd;if(!selected||selected.dataset.type!=='dimension')drawSelection()}}
 if(dimDraft&&tool==='dimension'){dimDraft.p2=p;dimPreview?.setAttribute('x2',p.x);dimPreview?.setAttribute('y2',p.y);return}
 if(textDraft&&tool==='text'){textDraft.p2=p;const x=Math.min(textDraft.p1.x,p.x),y=Math.min(textDraft.p1.y,p.y),w=Math.abs(p.x-textDraft.p1.x),h=Math.abs(p.y-textDraft.p1.y);for(const[k,v]of Object.entries({x,y,width:w,height:h}))textPreview?.setAttribute(k,v);return}
