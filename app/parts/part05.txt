function dimGeom(g){const p1=parsePt(g.dataset.p1),p2=parsePt(g.dataset.p2),vx=p2.x-p1.x,vy=p2.y-p1.y,L=Math.hypot(vx,vy)||1,ux=vx/L,uy=vy/L,nx=-uy,ny=ux,ts=+g.dataset.tailSize||7;return{p1,p2,ux,uy,nx,ny,ts}}
function drawDimHandles(g,hoverOnly=false){if(!g||g.dataset.type!=='dimension')return;const{p1,p2,nx,ny,ts}=dimGeom(g);[[p1,'start'],[p2,'end']].forEach(([p,k])=>{const lower={x:p.x-nx*ts/2,y:p.y-ny*ts/2},upper={x:p.x+nx*ts/2,y:p.y+ny*ts/2};paper.appendChild(el('line',{x1:lower.x,y1:lower.y,x2:upper.x,y2:upper.y,class:'dim-hover-line selection-ui'}));paper.appendChild(el('circle',{cx:lower.x,cy:lower.y,r:hoverOnly?.62:.72,class:`dim-end-handle dim-handle-${k} selection-ui`,'data-owner':g.dataset.id}));paper.appendChild(el('circle',{cx:upper.x,cy:upper.y,r:hoverOnly?.50:.58,class:`dim-tail-handle dim-tail-${k} selection-ui`,'data-owner':g.dataset.id}))})}
function drawTextHandles(g){const x=+g.dataset.x,y=+g.dataset.y,w=+g.dataset.w,h=+g.dataset.h,a=+g.dataset.angle||0,cx=x+w/2,cy=y+h/2;const grp=el('g',{class:'selection-ui',transform:`rotate(${a} ${cx} ${cy})`});grp.appendChild(el('rect',{x,y,width:w,height:h,rx:2.5,ry:2.5,class:'text-selection'}));grp.appendChild(el('rect',{x:x+w-1.7,y:y+h-1.7,width:3.4,height:3.4,rx:.7,ry:.7,class:'resize-handle'}));grp.appendChild(el('line',{x1:cx,y1:y,x2:cx,y2:y-8,class:'rotate-stem'}));grp.appendChild(el('circle',{cx,cy:y-8,r:1.5,class:'rotate-handle'}));paper.appendChild(grp)}
function drawSelection(){clearUI();if(!selected){if(hoverDim)drawDimHandles(hoverDim,true);return}if(selected.dataset.type==='dimension'){drawDimHandles(selected,false);return}if(selected.dataset.type==='text'){drawTextHandles(selected);return}if(selected.dataset.type==='image'){drawImageHandles(selected);return}}
function showProps(){const t=selected?.dataset.type;$('noSelection').hidden=!!t;$('imageProps').hidden=t!=='image';if(t==='dimension'){syncDimProps();openDimPopup();closeTextPopup()}else closeDimPopup();if(t==='text'){syncTextProps();openTextPopup();closeDimPopup()}else closeTextPopup();if(t==='image'){const r=imageRect(selected);$('imageOpacity').value=selected.getAttribute('opacity')||1;$('imageW').value=r.w.toFixed(1);$('imageH').value=r.h.toFixed(1)}}
function appendContent(n){const frame=paper.querySelector('[data-type=frame]');if(frame)paper.insertBefore(n,frame);else paper.appendChild(n)}
function createDim(p1,p2){const d=dimDefaults,g=el('g',{'data-type':'dimension','data-id':uid++,'data-p1':`${p1.x},${p1.y}`,'data-p2':`${p2.x},${p2.y}`,'data-value':'','data-prefix':d.prefix,'data-suffix':d.suffix,'data-arrow':d.arrow,'data-arrow-size':d.arrowSize,'data-arrow-angle':d.arrowAngle,'data-tail-size':d.tailSize,'data-line-width':d.lineWidth,'data-line-color':d.lineColor,'data-font':d.font,'data-font-size':d.fontSize,'data-text-color':d.textColor,'data-bold':d.bold?'1':'0','data-italic':d.italic?'1':'0'});appendContent(g);renderDim(g);select(g);setTool('dimension-edit',true);setTimeout(()=>{$('dimText').focus();$('dimText').select()},0)}
function drawArrow(g,p,ux,uy,s,angle,c,lw,dir,style){const dx=ux*dir,dy=uy*dir,nx=-dy,ny=dx;if(style==='dot'){g.appendChild(el('circle',{cx:p.x,cy:p.y,r:s*.20,fill:c}));return}if(style==='slash'){const a={x:p.x+nx*s*.46-dx*s*.16,y:p.y+ny*s*.46-dy*s*.16},b={x:p.x-nx*s*.46+dx*s*.16,y:p.y-ny*s*.46+dy*s*.16};g.appendChild(el('line',{x1:a.x,y1:a.y,x2:b.x,y2:b.y,stroke:c,'stroke-width':lw,'vector-effect':'non-scaling-stroke'}));return}let wf=1;if(style==='needle')wf=.32;else if(style==='slim')wf=.55;const half=Math.tan(angle*Math.PI/180)*s*wf,back={x:p.x+dx*s,y:p.y+dy*s},a={x:back.x+nx*half,y:back.y+ny*half},b={x:back.x-nx*half,y:back.y-ny*half};const closed=['closed','slim','needle'].includes(style);g.appendChild(el('path',{d:closed?`M ${p.x} ${p.y} L ${a.x} ${a.y} L ${b.x} ${b.y} Z`:`M ${a.x} ${a.y} L ${p.x} ${p.y} L ${b.x} ${b.y}`,fill:closed?c:'none',stroke:c,'stroke-width':Math.min(lw,.24),'stroke-linejoin':'miter','vector-effect':'non-scaling-stroke'}))}
function drawEndTail(g,p,ux,uy,h,c,lw){const nx=-uy,ny=ux,a={x:p.x-nx*h/2,y:p.y-ny*h/2},b={x:p.x+nx*h/2,y:p.y+ny*h/2};g.appendChild(el('line',{x1:a.x,y1:a.y,x2:b.x,y2:b.y,stroke:c,'stroke-width':lw,'vector-effect':'non-scaling-stroke'}))}
function renderDim(g){while(g.firstChild)g.removeChild(g.firstChild);const{p1,p2,ux,uy}=dimGeom(g),vx=p2.x-p1.x,vy=p2.y-p1.y,lw=+g.dataset.lineWidth||.4,c=g.dataset.lineColor||'#111',fs=+g.dataset.fontSize||4,as=+g.dataset.arrowSize||5,aa=+g.dataset.arrowAngle||10,ts=+g.dataset.tailSize||7;g.appendChild(el('line',{x1:p1.x,y1:p1.y,x2:p2.x,y2:p2.y,stroke:c,'stroke-width':lw,'vector-effect':'non-scaling-stroke'}));drawArrow(g,p1,ux,uy,as,aa,c,lw,1,g.dataset.arrow);drawArrow(g,p2,ux,uy,as,aa,c,lw,-1,g.dataset.arrow);drawEndTail(g,p1,ux,uy,ts,c,lw);drawEndTail(g,p2,ux,uy,ts,c,lw);const mid={x:(p1.x+p2.x)/2,y:(p1.y+p2.y)/2},ang=Math.atan2(vy,vx)*180/Math.PI;let ta=ang;if(ta>90||ta<-90)ta+=180;let nx=-uy,ny=ux;if(ny>0){nx=-nx;ny=-ny}const off=fs*.9+1.1,tm={x:mid.x+nx*off,y:mid.y+ny*off};const label=(g.dataset.prefix||'')+(g.dataset.value||'…')+(g.dataset.suffix||'');const t=el('text',{x:tm.x,y:tm.y,'text-anchor':'middle','dominant-baseline':'central','font-family':g.dataset.font||'Bahnschrift','font-size':fs,'font-weight':g.dataset.bold==='1'?'700':'400','font-style':g.dataset.italic==='1'?'italic':'normal',fill:g.dataset.textColor||c,transform:`rotate(${ta} ${tm.x} ${tm.y})`,class:'dim-label'});t.textContent=label;g.appendChild(t);g.appendChild(el('line',{x1:p1.x,y1:p1.y,x2:p2.x,y2:p2.y,class:'dim-hit'}))}
