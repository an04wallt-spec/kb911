function downloadBlob(blob,name){const url=URL.createObjectURL(blob),a=document.createElement('a');a.href=url;a.download=name;a.style.display='none';document.body.appendChild(a);a.click();setTimeout(()=>{a.remove();URL.revokeObjectURL(url)},10000)}
function roundedRectPath(ctx,x,y,w,h,r){r=Math.max(0,Math.min(r,w/2,h/2));ctx.beginPath();ctx.moveTo(x+r,y);ctx.arcTo(x+w,y,x+w,y+h,r);ctx.arcTo(x+w,y+h,x,y+h,r);ctx.arcTo(x,y+h,x,y,r);ctx.arcTo(x,y,x+w,y,r);ctx.closePath()}
function parseColor(v,def='#111111'){return(!v||v==='none')?def:v}
function loadCanvasImage(src){return new Promise((resolve,reject)=>{const im=new Image();im.onload=()=>resolve(im);im.onerror=()=>reject(new Error('Не удалось прочитать изображение на листе'));im.src=src})}
async function drawImageObject(ctx,n){const href=n.getAttribute('href')||n.getAttributeNS('http://www.w3.org/1999/xlink','href');if(!href)return;const im=await loadCanvasImage(href);ctx.save();ctx.globalAlpha=+(n.getAttribute('opacity')||1);ctx.drawImage(im,+n.getAttribute('x'),+n.getAttribute('y'),+n.getAttribute('width'),+n.getAttribute('height'));ctx.restore()}
function applyRotateTransform(ctx,tr){if(!tr)return;const m=/rotate\(\s*(-?[\d.]+)(?:[ ,]+(-?[\d.]+)[ ,]+(-?[\d.]+))?\s*\)/.exec(tr);if(!m)return;const a=+m[1]*Math.PI/180,cx=+(m[2]||0),cy=+(m[3]||0);ctx.translate(cx,cy);ctx.rotate(a);ctx.translate(-cx,-cy)}
function drawSvgLine(ctx,n){if(n.classList.contains('dim-hit'))return;ctx.save();ctx.strokeStyle=parseColor(n.getAttribute('stroke'));ctx.lineWidth=+(n.getAttribute('stroke-width')||.2);ctx.beginPath();ctx.moveTo(+n.getAttribute('x1'),+n.getAttribute('y1'));ctx.lineTo(+n.getAttribute('x2'),+n.getAttribute('y2'));ctx.stroke();ctx.restore()}
function drawSvgCircle(ctx,n){ctx.save();const fill=n.getAttribute('fill'),stroke=n.getAttribute('stroke');if(fill&&fill!=='none'){ctx.fillStyle=fill;ctx.beginPath();ctx.arc(+n.getAttribute('cx'),+n.getAttribute('cy'),+n.getAttribute('r'),0,Math.PI*2);ctx.fill()}if(stroke&&stroke!=='none'){ctx.strokeStyle=stroke;ctx.lineWidth=+(n.getAttribute('stroke-width')||.2);ctx.beginPath();ctx.arc(+n.getAttribute('cx'),+n.getAttribute('cy'),+n.getAttribute('r'),0,Math.PI*2);ctx.stroke()}ctx.restore()}
function drawSvgPath(ctx,n){ctx.save();const path=new Path2D(n.getAttribute('d')||'');const fill=n.getAttribute('fill'),stroke=n.getAttribute('stroke');if(fill&&fill!=='none'){ctx.fillStyle=fill;ctx.fill(path)}if(stroke&&stroke!=='none'){ctx.strokeStyle=stroke;ctx.lineWidth=+(n.getAttribute('stroke-width')||.2);ctx.lineJoin=n.getAttribute('stroke-linejoin')||'miter';ctx.stroke(path)}ctx.restore()}
function drawSvgText(ctx,n){ctx.save();applyRotateTransform(ctx,n.getAttribute('transform'));const fs=+(n.getAttribute('font-size')||4),fam=n.getAttribute('font-family')||'Bahnschrift',fw=n.getAttribute('font-weight')||'400',fst=n.getAttribute('font-style')||'normal';ctx.font=`${fst} ${fw} ${fs}px ${fam}`;ctx.fillStyle=n.getAttribute('fill')||'#111';ctx.textAlign=n.getAttribute('text-anchor')==='middle'?'center':n.getAttribute('text-anchor')==='end'?'right':'left';ctx.textBaseline=n.getAttribute('dominant-baseline')==='central'?'middle':'alphabetic';const baseX=+(n.getAttribute('x')||0),baseY=+(n.getAttribute('y')||0);const spans=[...n.children].filter(x=>x.tagName.toLowerCase()==='tspan');if(!spans.length){ctx.fillText(n.textContent||'',baseX,baseY)}else{let y=baseY;for(const sp of spans){const dy=+(sp.getAttribute('dy')||0);y+=dy;ctx.fillText(sp.textContent||'',+(sp.getAttribute('x')||baseX),y)}}ctx.restore()}
function drawDimensionObject(ctx,g){ctx.save();applyRotateTransform(ctx,g.getAttribute('transform'));for(const n of g.children){const tag=n.tagName.toLowerCase();if(tag==='line')drawSvgLine(ctx,n);else if(tag==='path')drawSvgPath(ctx,n);else if(tag==='circle')drawSvgCircle(ctx,n);else if(tag==='text')drawSvgText(ctx,n)}ctx.restore()}
function drawTextObject(ctx,g){const x=+g.dataset.x,y=+g.dataset.y,w=+g.dataset.w,h=+g.dataset.h,fs=+g.dataset.fontSize||4,font=g.dataset.font||'Bahnschrift',ang=+g.dataset.angle||0,align=g.dataset.align||'center',cx=x+w/2,cy=y+h/2;ctx.save();ctx.translate(cx,cy);ctx.rotate(ang*Math.PI/180);ctx.translate(-cx,-cy);const opa=+(g.dataset.bgOpacity||0);if(opa>0){ctx.globalAlpha=opa;ctx.fillStyle=g.dataset.bg||'#fff';roundedRectPath(ctx,x,y,w,h,2.5);ctx.fill();ctx.globalAlpha=1}ctx.beginPath();roundedRectPath(ctx,x+1,y+1,Math.max(0,w-2),Math.max(0,h-2),2);ctx.clip();ctx.fillStyle=g.dataset.textColor||'#111';ctx.font=`${g.dataset.bold==='1'?'700':'400'} ${fs}px ${font}`;ctx.textAlign=align==='left'?'left':align==='right'?'right':'center';ctx.textBaseline='middle';const tx=align==='left'?x+2:align==='right'?x+w-2:x+w/2,lines=wrapTextLines(g.dataset.value||'',font,fs,Math.max(2,w-4),g.dataset.bold==='1'),lh=fs*1.22,totalH=lines.length?fs+(lines.length-1)*lh:fs,firstY=cy-totalH/2+fs/2;lines.forEach((line,i)=>ctx.fillText(line,tx,firstY+i*lh));ctx.restore()}
